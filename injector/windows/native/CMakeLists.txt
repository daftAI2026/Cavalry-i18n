# [INPUT]: Qt6.9.2 x64 SDK、官方MetaData/appStrings/Presets、共享说明/缓动/五项菜单/Demo producer及Windows QPA/CAS。
# [OUTPUT]: native generic/QPA、显示策略合同及官方PE身份/节点绘制调用点证明；不复制Qt。
# [POS]: Windows2.8.1专用窄运行时构建；共享producer生成翻译，独立Windows ABI合同约束接线。
# [PROTOCOL]: 变更时更新此头部，然后检查 CLAUDE.md
cmake_minimum_required(VERSION 4.2)
project(cavalry_native_windows LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Cavalry2.8 native runtime requires Windows MSVC x64")
endif()
find_package(Qt6 6.9.2 EXACT REQUIRED COMPONENTS Core Gui Widgets)
foreach(target Qt6::Core Qt6::Gui Qt6::Widgets)
  get_target_property(kind ${target} TYPE)
  if(NOT kind STREQUAL "SHARED_LIBRARY")
    message(FATAL_ERROR "Use only the selected host's shared Qt runtime")
  endif()
endforeach()

qt_standard_project_setup()
add_compile_options(/utf-8)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(NATIVE_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(CAVALRY_NATIVE_METADATA_DIR "" CACHE PATH "Selected official Cavalry2.8 assets/MetaData directory")
if(NOT IS_DIRECTORY "${CAVALRY_NATIVE_METADATA_DIR}")
  message(FATAL_ERROR "Set CAVALRY_NATIVE_METADATA_DIR to the selected official Cavalry2.8 assets/MetaData directory")
endif()
find_program(NATIVE_NODE_EXECUTABLE node REQUIRED)
string(SHA256 nativeBuildKey "${CMAKE_BINARY_DIR}")
string(SUBSTRING "${nativeBuildKey}" 0 12 nativeBuildKey)
file(TO_CMAKE_PATH "$ENV{TEMP}" nativeTempRoot)
set(nativeGenerated "${nativeTempRoot}/cavalry-native-windows-${nativeBuildKey}")
file(MAKE_DIRECTORY "${nativeGenerated}")
execute_process(COMMAND "${NATIVE_NODE_EXECUTABLE}" "${NATIVE_REPO_ROOT}/tools/build_native_help_description_table.js"
  --metadata-dir "${CAVALRY_NATIVE_METADATA_DIR}" --out "${nativeGenerated}/help-descriptions.inc" --cavalry-version 2.8.1
  RESULT_VARIABLE helpResult OUTPUT_VARIABLE helpOutput ERROR_VARIABLE helpError)
if(NOT helpResult EQUAL 0)
  message(FATAL_ERROR "Official help projection failed: ${helpError}")
endif()
execute_process(COMMAND "${NATIVE_NODE_EXECUTABLE}" "${NATIVE_REPO_ROOT}/tools/build_native_easing_display_table.js"
  --out "${nativeGenerated}/easing.inc"
  RESULT_VARIABLE easingResult OUTPUT_VARIABLE easingOutput ERROR_VARIABLE easingError)
if(NOT easingResult EQUAL 0)
  message(FATAL_ERROR "Approved easing projection failed: ${easingError}")
endif()
get_filename_component(nativeAssetsRoot "${CAVALRY_NATIVE_METADATA_DIR}/.." ABSOLUTE)
execute_process(COMMAND "${NATIVE_NODE_EXECUTABLE}" "${NATIVE_REPO_ROOT}/tools/build_native_dg_context_menu_table.js"
  --assets-dir "${nativeAssetsRoot}" --out "${nativeGenerated}/dg-context-menu.inc"
  RESULT_VARIABLE dgResult ERROR_VARIABLE dgError)
if(NOT dgResult EQUAL 0)
  message(FATAL_ERROR "Approved context-menu projection failed: ${dgError}")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${NATIVE_REPO_ROOT}/tools/build_native_dg_context_menu_table.js"
  "${NATIVE_REPO_ROOT}/tools/generate_embedded_translations.js"
  "${NATIVE_REPO_ROOT}/tools/native_dg_context_menu_data.js"
  "${NATIVE_REPO_ROOT}/tools/zh-Hans.ts" "${NATIVE_REPO_ROOT}/tools/zh-Hant.ts" "${NATIVE_REPO_ROOT}/tools/ja_JP.ts"
  "${nativeAssetsRoot}/Definitions/Translation/appStrings_en-GB.source.json"
  "${nativeAssetsRoot}/Definitions/Translation/appStrings_ja-JP.translation.json")
execute_process(COMMAND "${NATIVE_NODE_EXECUTABLE}" "${NATIVE_REPO_ROOT}/tools/native_windows_catalog_warning_guard.js"
  --assets-root "${nativeAssetsRoot}" --out "${nativeGenerated}/catalog-warning-guard.inc"
  RESULT_VARIABLE catalogWarningResult ERROR_VARIABLE catalogWarningError)
if(NOT catalogWarningResult EQUAL 0)
  message(FATAL_ERROR "Approved catalog warning guard failed: ${catalogWarningError}")
endif()
execute_process(COMMAND "${NATIVE_NODE_EXECUTABLE}" "${NATIVE_REPO_ROOT}/tools/generate_demo_reader_tables.js"
  --source "${NATIVE_REPO_ROOT}/languages/native/official_demo_scripts.js" --presets-root "${nativeAssetsRoot}/Presets"
  --source-line-endings crlf --texts-out "${nativeGenerated}/demo-texts.inc" --files-out "${nativeGenerated}/demo-files.inc"
  RESULT_VARIABLE demoResult OUTPUT_VARIABLE demoOutput ERROR_VARIABLE demoError)
if(NOT demoResult EQUAL 0)
  message(FATAL_ERROR "Official Windows Demo projection failed: ${demoError}")
endif()
file(GLOB helpInputs CONFIGURE_DEPENDS "${NATIVE_REPO_ROOT}/tools/native_*.js" "${NATIVE_REPO_ROOT}/languages/native/*.js" "${NATIVE_REPO_ROOT}/languages/*/MetaData/*.json" "${CAVALRY_NATIVE_METADATA_DIR}/*.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${helpInputs} "${NATIVE_REPO_ROOT}/tools/build_native_help_description_table.js" "${NATIVE_REPO_ROOT}/tools/build_native_easing_display_table.js" "${NATIVE_REPO_ROOT}/tools/native_windows_catalog_warning_guard.js" "${NATIVE_REPO_ROOT}/tools/cavalry_native_target.json")
file(GLOB_RECURSE demoInputs CONFIGURE_DEPENDS "${nativeAssetsRoot}/Presets/*.js")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${demoInputs} "${NATIVE_REPO_ROOT}/tools/generate_demo_reader_tables.js")
qt_add_plugin(cavalryi18n SHARED CLASS_NAME CavalryI18nNativePlugin)
target_sources(cavalryi18n PRIVATE plugin.cpp runtime.cpp runtime.h ../cavalry_i18n_iat_patch.cpp)
target_link_libraries(cavalryi18n PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets shell32 ole32 uuid)
target_include_directories(cavalryi18n PRIVATE "${NATIVE_REPO_ROOT}/injector/native")
target_compile_definitions(cavalryi18n PRIVATE NATIVE_HELP_DESCRIPTION_TABLE="${nativeGenerated}/help-descriptions.inc")
target_compile_definitions(cavalryi18n PRIVATE NATIVE_HELP_LAYOUT_ROWS="${nativeGenerated}/help-descriptions.inc.layout.inc")
target_compile_definitions(cavalryi18n PRIVATE NATIVE_CATALOG_WARNING_GUARD="${nativeGenerated}/catalog-warning-guard.inc")
target_compile_definitions(cavalryi18n PRIVATE NATIVE_EASING_DISPLAY_TABLE="${nativeGenerated}/easing.inc")
target_compile_definitions(cavalryi18n PRIVATE NATIVE_DG_CONTEXT_MENU_TABLE="${nativeGenerated}/dg-context-menu.inc")
target_compile_definitions(cavalryi18n PRIVATE NATIVE_DEMO_READER_TEXTS_TABLE="${nativeGenerated}/demo-texts.inc" NATIVE_DEMO_READER_FILES_TABLE="${nativeGenerated}/demo-files.inc")
target_compile_definitions(cavalryi18n PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX QT_NO_KEYWORDS)
qt_add_plugin(cavalryi18n_qpa SHARED CLASS_NAME CavalryWindowsQpaProxy)
target_sources(cavalryi18n_qpa PRIVATE ../cavalry_i18n_qpa_proxy.cpp ../cavalry_i18n_qpa_proxy.h ../cavalry_i18n_qpa_contract.cpp ../cavalry_i18n_qpa_contract.h ../qwindows.json)
target_link_libraries(cavalryi18n_qpa PRIVATE Qt6::Core Qt6::Gui Qt6::GuiPrivate)
target_compile_definitions(cavalryi18n_qpa PRIVATE CAVALRY_I18N_NATIVE_28 QT_NO_KEYWORDS)
set_target_properties(cavalryi18n PROPERTIES OUTPUT_NAME cavalryi18n PREFIX "")
set_target_properties(cavalryi18n_qpa PROPERTIES OUTPUT_NAME qwindows PREFIX "")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(cavalryi18n PROPERTIES "RUNTIME_OUTPUT_DIRECTORY_${config}" "${CMAKE_BINARY_DIR}/generic" "LIBRARY_OUTPUT_DIRECTORY_${config}" "${CMAKE_BINARY_DIR}/generic")
  set_target_properties(cavalryi18n_qpa PROPERTIES "RUNTIME_OUTPUT_DIRECTORY_${config}" "${CMAKE_BINARY_DIR}/qpa" "LIBRARY_OUTPUT_DIRECTORY_${config}" "${CMAKE_BINARY_DIR}/qpa")
endforeach()
